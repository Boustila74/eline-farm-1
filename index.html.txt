  <script>
    const display = document.getElementById('display');
    let currentInput = '0';
    let storedValue = null;
    let operation = null;
    let startNewInput = false;

    function updateDisplay() {
      display.textContent = currentInput;
    }

    function enterNumber(number) {
      if (startNewInput || currentInput === 'Error') {
        currentInput = number;
        startNewInput = false;
      } else if (currentInput === '0') {
        currentInput = number;
      } else {
        currentInput += number;
      }
      updateDisplay();
    }

    function enterDecimal() {
      if (startNewInput || currentInput === 'Error') {
        currentInput = '0';
        startNewInput = false;
      }
      if (!currentInput.includes('.')) currentInput += '.';
      updateDisplay();
    }

    function calculate() {
      if (operation === null || storedValue === null) return;

      const rightValue = Number(currentInput);
      let result;

      switch (operation) {
        case '+': result = storedValue + rightValue; break;
        case '-': result = storedValue - rightValue; break;
        case '*': result = storedValue * rightValue; break;
        case '/': result = rightValue === 0 ? NaN : storedValue / rightValue; break;
        default: return;
      }

      currentInput = Number.isFinite(result)
        ? String(Number(result.toPrecision(10)))
        : 'Error';
      storedValue = null;
      operation = null;
      startNewInput = true;
      updateDisplay();
    }

    function chooseOperation(nextOperation) {
      if (currentInput === 'Error') return;

      if (operation !== null && !startNewInput) {
        calculate();
        if (currentInput === 'Error') return;
      }

      storedValue = Number(currentInput);
      operation = nextOperation;
      startNewInput = true;
    }

    function clearCalculator() {
      currentInput = '0';
      storedValue = null;
      operation = null;
      startNewInput = false;
      updateDisplay();
    }

    function deleteDigit() {
      if (startNewInput || currentInput === 'Error') {
        currentInput = '0';
        startNewInput = false;
      } else {
        currentInput = currentInput.length > 1
          ? currentInput.slice(0, -1)
          : '0';
        if (currentInput === '-') currentInput = '0';
      }
      updateDisplay();
    }

    document.querySelector('.buttons').addEventListener('click', (event) => {
      const button = event.target.closest('button');
      if (!button) return;

      if (button.dataset.number !== undefined) {
        enterNumber(button.dataset.number);
      } else if (button.dataset.operation) {
        chooseOperation(button.dataset.operation);
      } else {
        switch (button.dataset.action) {
          case 'decimal': enterDecimal(); break;
          case 'equals': calculate(); break;
          case 'clear': clearCalculator(); break;
          case 'delete': deleteDigit(); break;
        }
      }
    });

    document.addEventListener('keydown', (event) => {
      if (/^[0-9]$/.test(event.key)) enterNumber(event.key);
      else if (event.key === '.') enterDecimal();
      else if (['+', '-', '*', '/'].includes(event.key)) chooseOperation(event.key);
      else if (event.key === 'Enter' || event.key === '=') {
        event.preventDefault();
        calculate();
      } else if (event.key === 'Backspace') deleteDigit();
      else if (event.key === 'Escape') clearCalculator();
    });
  </script>
